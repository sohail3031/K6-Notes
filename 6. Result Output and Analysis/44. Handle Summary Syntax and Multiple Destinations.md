```javascript
import http from "k6/http";
import { textSummary } from "https://jslib.k6.io/k6-summary/0.0.1/index.js";

export default function() {
	http.get("https://test.k6.io");
}

export function handleSummary(data) {
	return {
		"stdout": JSON.stringify(data, null, 2), // print data on standard output
		"summary.json": JSON.stringify(data), // safe data in json file
		"summary.html": generateHTMLSummary(data)
	};
}

function generateHTMLSummary(data) {
	let html = `<html><body><h1>Text Summary</h1>`;
	html += `<pre>${JSON.stringify(data, null, 2)}</pre>`;
	html += `</body></html>`;
	
	return html;
}
```
