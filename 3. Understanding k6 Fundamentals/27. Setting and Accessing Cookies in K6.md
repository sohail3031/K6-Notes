Cookies are small piece of information that is stored in the client side and send to the server when a request is made. Lets see some example on how to use cookies in K6.

**Code:** 

```javascript
import http from "k6/http";
import { check, sleep } from "k6";

export default function () {
    const res = http.get("https://quickpizza.grafana.com/api/cookies", {
        headers: {
            Cookie: "my_cookie=hello world"
        }
    });
    const body = res.json();

    check(res, {
        "status is 200":            (r) => r.status === 200,
        "has cookie 'my_cookie'":   () => body.cookies.my_cookie !== undefined,
        "cookie has correct value": () => body.cookies.my_cookie === "hello world"
    });

    sleep(1);
}
```

**Code:**

```javascript
import http from "k6/http";
import { check, sleep } from "k6";

export default function() {
	const jar = http.cookieJar();
	
	jar.set("https://quickpizza.grafana.com/api/cookies", "my_cookie", "hello world");
	
	const cookies = {
		my_cookie: {
			value: "hello world 2",
			replace: true
		}
	};
	const res = http.get("https://quickpizza.grafana.com/api/cookies", { cookies });
	
	check(res.json(), {
		"cookie has correct value": (r) => r.cookies.my_cookie == "hello world 2"
	});
	
	sleep(1);
}
```
