In the following code snippets, we are going to see how to add threshold conditions on custom metrics.

**Code:**

```javascript
import { Counter } from "k6/metrics";
import { sleep } from "k6";

let myCounter = new Counter("my_counter");

export let options = {
	thresholds: {
		"my_counter": ["count>100"]
	},
	vus: 1,
	duration: "10s"
}

export default function() {
	myCounter.add(1); // increment the counter by 1 for each iteration
	
	sleep(0.1);
}
```

**Code:**

```javascript
import http from "k6/http";
import { Rate } from "k6/metrics";
import { sleep } from "k6";

let myRate = new Rate("my_rate");

export let options = {
	thresholds: {
		"my_rate": ["rate>0.95"]
	},
	vus: 1,
	duration: "10s"
}

export default function() {
	let res = http.get("https://test-api.k6.io/public/crocodiles/1/");
	
	myRate.add(res.status === 200);
	
	sleep(0.1);
}
```
