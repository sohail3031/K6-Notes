# Why use Tags

1. **Filtering Metrics:** Tags help you filter and categories metrics, making it easier to focus on specific parts of your test.
2. **Detailed Analysis:** They allow for more granular analysis, helping you identify issues related to specific scenarios or endpoints.
3. **Organizing Results:** Tags help in organizing results when dealing with large and complex text scripts.

**Tags Code:**

```javascript
import http from "k6/http";
import { check, sleep } from "k6";

let params = {
	tags: {name: "homepage"}
}

export default function() {
	let res = http.get("https://test.k6.io/", params);
	let tags = params.tags;
	
	check(res, {
		[`is status 200 for ${tags.name}`]: (r) => r.status === 200,
		[`response time for ${tags.name} is below 200ms`]: (r) => r.timings.duration < 200
	});
	
	console.log(`Tags: ${JSON.stringify(tags)}, Status: ${res.status}`);
	
	sleep(1);
}
```

**Group Code:**

```javascript
import http from "k6/http";
import { check, group, sleep } from "k6";

export default function() {
	group("Home Page", function() {
		let res = http.get("https://test.k6.io/");
		
		check(res, {
			"status is 200": (r) => r.status === 200
		});
		
		sleep(1);
	});
	
	group("Contact Page", function() {
		let res = http.get("https://test.k6.io/contact");
		
		check(res, {
			"status is 200": (r) => r.status === 200
		});
		
		sleep(1);
	});
}
```

**Group Code:**

```javascript
import http from "k6/http";
import { check, group, sleep } from "k6";
import { Counter } from "k6/metrics";

let httpReqs = new Counter("http_reqs");

export default function() {
	group("Home Page", function() {
		let params = {
			tags: {name: "homePage"}
		};
		let res = http.get("https://test.k6.io/", params);
		let tags = params.tags;
		
		check(res, {
			[`is status is 200 for ${tags.name}`]: (r) => r.status === 200
		});
		
		httpReqs.add(1, tags);
		
		sleep(1);
	});
	
	group("Contact Page", function() {
		let params = {
			tags: {name: "contactPage"}
		};
		let res = http.get("https://test.k6.io/contact", params);
		let tags = params.tags;
		
		check(res, {
			[`is status is 200 for ${tags.name}`]: (r) => r.status === 200
		});
		
		httpReqs.add(1, tags);
		
		sleep(1);
	});
}
```
