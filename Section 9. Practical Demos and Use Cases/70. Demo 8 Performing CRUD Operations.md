## Utility Function: Random String Generator

```javascript
function randomString(length, charset = "") {
    if (!charset) {
        charset = "abcdefghijklmnopqrstuvwxyz";
    }

    let res = "";

    while (length--) {
        res += charset[(Math.random() * charset.length) | 0];
    }

    return res;
}
```

### Purpose

- Generates random usernames
- Simulates realistic test data
- Used for user registration

## Setup Function (User Creation + Login)

The `setup()` function runs once before the test.

### 1. Register User

```javascript
export function setup() {
    const registerRes = http.post(
        `${BASE_URL}/api/users`,
        JSON.stringify({
            username: USERNAME,
            password: PASSWORD,
        }),
        JSON_HEADERS
    );

    check(registerRes, {
        "Created User": (r) => r.status === 201,
    });
```

---

### 2. Login User

```javascript
    const loginRes = http.post(
        `${BASE_URL}/api/users/token/login`,
        JSON.stringify({
            username: USERNAME,
            password: PASSWORD,
        }),
        JSON_HEADERS
    );

    const userToken = loginRes.json("token");

    check(userToken, {
        "Logged in successfully": () => !!userToken,
    });

    return { userToken };
}
```

### Explanation

- Creates user account
- Logs in using credentials
- Retrieves authentication token
- Passes token to main function

## 1. CREATE Operation

### Get Resource (Pizza Recommendation)

```javascript
let pizzaId = null;

group("01. Get a pizza recommendation", () => {
    const pizzaRes = http.post(
        `${BASE_URL}/api/pizza`,
        JSON.stringify({
            maxCaloriesPerSlice: 500,
            mustBeVegetarian: false,
            excludedIngredients: [],
            excludedTools: [],
            maxNumberOfToppings: 6,
            minNumberOfToppings: 2,
        }),
        userConfig({ name: "GetPizza" })
    );

    if (
        !check(pizzaRes, {
            "Got pizza recommendation": (r) => r.status === 200,
        })
    ) {
        console.log(`Failed to get pizza: ${pizzaRes.status} ${pizzaRes.body}`);
        return;
    }

    pizzaId = pizzaRes.json("pizza.id");
});
```

---

## 2. CREATE (Rating Resource)

```javascript
let createdRatingId = null;

if (pizzaId) {
    const ratingRes = http.post(
        `${BASE_URL}/api/ratings`,
        JSON.stringify({
            stars: 4,
            pizza_id: pizzaId,
        }),
        userConfig({ name: "RatePizza" })
    );

    if (check(ratingRes, { "Rating created": (r) => r.status === 201 })) {
        createdRatingId = ratingRes.json("id");
    } else {
        console.log(
            `Failed to create rating: ${ratingRes.status} ${ratingRes.body}`
        );
    }
}
```

---

## 3. READ Operation

```javascript
group("02. Fetch my ratings", () => {
    const res = http.get(
        `${BASE_URL}/api/ratings`,
        userConfig({ name: "Fetch" })
    );

    check(res, {
        "Retrieved ratings status": (r) => r.status === 200,
    });

    check(res.json(), {
        "Retrieved ratings list": (r) => {
            const list = r.ratings ?? r.results ?? r.data ?? r;
            return Array.isArray(list) && list.length > 0;
        },
    });
});
```

---

## 4. UPDATE Operation

```javascript
group("03. Update a rating", () => {
    if (!createdRatingId) {
        console.log("No rating ID to update, skipping");
        return;
    }

    const res = http.patch(
        `${BASE_URL}/api/ratings/${createdRatingId}`,
        JSON.stringify({ stars: 5 }),
        userConfig({ name: "Update" })
    );

    check(res, {
        "Update endpoint reachable": (r) => r.status !== 0,
    });

    if (res.status !== 200) {
        console.log(`Update returned: ${res.status} ${res.body}`);
    }
});
```

---

## 5. DELETE Operation

```javascript
group("04. Delete a rating", () => {
    if (!createdRatingId) {
        console.log("No rating ID to delete, skipping");
        return;
    }

    const res = http.del(
        `${BASE_URL}/api/ratings/${createdRatingId}`,
        null,
        userConfig({ name: "Delete" })
    );

    check(res, {
        "Delete endpoint reachable": (r) => r.status !== 0,
    });

    if (res.status !== 204) {
        console.log(`Delete returned: ${res.status} ${res.body}`);
    }
});
```
