K6 offers us to perform CRUD operations on a database. See the below code on how to do it.

**Code:**

```javascript
import http from "k6/http";
import { check, group } from "k6";

export const options = {
  vus: 1,
  iterations: 1,
};

function randomString(length, charset = "") {
  if (!charset) {
	  charset = "abcdefghijklmnopqrstuvwxyz";
  }

  let res = "";

  while (length--) {
	  res += charset[(Math.random() * charset.length) | 0];
  }

  return res;
}

const USERNAME = `${randomString(10)}@example.com`;
const PASSWORD = "secret";
const BASE_URL = "https://quickpizza.grafana.com";
const JSON_HEADERS = { headers: { "Content-Type": "application/json" } };

export function setup() {
  const registerRes = http.post(
    `${BASE_URL}/api/users`,
    JSON.stringify({ username: USERNAME, password: PASSWORD }),
    JSON_HEADERS
  );

  check(registerRes, { "Created User": (r) => r.status === 201 });

  const loginRes = http.post(
    `${BASE_URL}/api/users/token/login`,
    JSON.stringify({ username: USERNAME, password: PASSWORD }),
    JSON_HEADERS
  );

  const userToken = loginRes.json("token");
  
  check(userToken, { "Logged in successfully": () => !!userToken });

  return { userToken };
}

export default ({ userToken }) => {
  const userConfig = (tag) => ({
    headers: {
      Authorization: `Token ${userToken}`,
      "Content-Type": "application/json",
    },
    tags: Object.assign({}, { name: "PrivatePizza" }, tag),
  });

  let pizzaId = null;

  group("01. Get a pizza recommendation", () => {
    const pizzaRes = http.post(
      `${BASE_URL}/api/pizza`,
      JSON.stringify({
        maxCaloriesPerSlice: 500,
        mustBeVegetarian: false,
        excludedIngredients: [],
        excludedTools: [],
        maxNumberOfToppings: 6,
        minNumberOfToppings: 2,
      }),
      userConfig({ name: "GetPizza" })
    );

    if (!check(pizzaRes, { "Got pizza recommendation": (r) => r.status === 200 })) {
      console.log(`Failed to get pizza: ${pizzaRes.status} ${pizzaRes.body}`);

      return;
    }

    pizzaId = pizzaRes.json("pizza.id");
  });

  let createdRatingId = null;

  if (pizzaId) {
    const ratingRes = http.post(
      `${BASE_URL}/api/ratings`,
      JSON.stringify({ stars: 4, pizza_id: pizzaId }),
      userConfig({ name: "RatePizza" })
    );

    if (check(ratingRes, { "Rating created": (r) => r.status === 201 })) {
      createdRatingId = ratingRes.json("id");
    } else {
      console.log(`Failed to create rating: ${ratingRes.status} ${ratingRes.body}`);
    }
  }

  group("02. Fetch my ratings", () => {
    const res = http.get(`${BASE_URL}/api/ratings`, userConfig({ name: "Fetch" }));

    check(res, { "Retrieved ratings status": (r) => r.status === 200 });
    check(res.json(), {
      "Retrieved ratings list": (r) => {
        const list = r.ratings ?? r.results ?? r.data ?? r;
        return Array.isArray(list) && list.length > 0;
      },
    });
  });

  group("03. Update a rating", () => {
    if (!createdRatingId) {
	    console.log("No rating ID to update, skipping"); 
	    
	    return; 
    }

    const res = http.patch(
      `${BASE_URL}/api/ratings/${createdRatingId}`,
      JSON.stringify({ stars: 5 }),
      userConfig({ name: "Update" })
    );

    check(res, {
      "Update endpoint reachable": (r) => r.status !== 0,
    });

    if (res.status !== 200) {
	    console.log(`Update returned: ${res.status} ${res.body}`);
    }
  });

  group("04. Delete a rating", () => {
    if (!createdRatingId) {
	    console.log("No rating ID to delete, skipping"); 
	    
	    return;
    }

    const res = http.del(
      `${BASE_URL}/api/ratings/${createdRatingId}`,
      null,
      userConfig({ name: "Delete" })  // fixed: was adminConfig
    );

    check(res, {
      "Delete endpoint reachable": (r) => r.status !== 0,
    });

    if (res.status !== 204) {
	    console.log(`Delete returned: ${res.status} ${res.body}`);
    }
  });
};
```
